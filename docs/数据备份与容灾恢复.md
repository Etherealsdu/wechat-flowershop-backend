## 备份策略概述

### 3-2-1 备份原则

```
┌─────────────────────────────────────────────────────────────┐
│                     3-2-1 备份原则                           │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│   3 份数据副本                                               │
│   ├── 1 份生产数据                                          │
│   ├── 1 份本地备份                                          │
│   └── 1 份异地备份                                          │
│                                                             │
│   2 种存储介质                                               │
│   ├── 本地磁盘/SSD                                          │
│   └── 云存储 (OSS/S3)                                       │
│                                                             │
│   1 份异地备份                                               │
│   └── 不同地域的云存储                                       │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

### 备份类型与频率

| 备份类型 | 频率 | 保留时间 | 说明 |
|----------|------|----------|------|
| 全量备份 | 每周日 02:00 | 30天 | 完整数据库备份 |
| 增量备份 | 每天 02:00 | 7天 | 基于 binlog 的增量备份 |
| 实时备份 | 实时 | 7天 | MySQL binlog 实时同步 |
| 配置备份 | 每周 | 90天 | 系统和应用配置文件 |
| 文件备份 | 每天 | 30天 | OSS 文件同步 |

### 备份存储位置

| 存储位置 | 用途 | 容量 |
|----------|------|------|
| `/data/backup/mysql` | MySQL 本地备份 | 100GB |
| `/data/backup/config` | 配置文件备份 | 10GB |
| `oss://backup-bucket` | 云端备份 | 不限 |
| `oss://dr-bucket` | 异地灾备 | 不限 |

## 数据库备份

### 全量备份脚本

```bash
#!/bin/bash
# /home/user/scripts/mysql_full_backup.sh
# MySQL 全量备份脚本

set -e

# 配置
BACKUP_DIR="/data/backup/mysql"
DATE=$(date +%Y%m%d_%H%M%S)
DB_NAME="wechat_flowershop"
DB_USER="backup_user"
DB_PASS="your_backup_password"
RETENTION_DAYS=30
LOG_FILE="${BACKUP_DIR}/backup.log"

# 函数：记录日志
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" >> ${LOG_FILE}
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1"
}

# 函数：发送告警
send_alert() {
    local message=$1
    # 钉钉告警
    if [ -n "$DINGTALK_WEBHOOK" ]; then
        curl -s -X POST "$DINGTALK_WEBHOOK" \
            -H "Content-Type: application/json" \
            -d "{\"msgtype\":\"text\",\"text\":{\"content\":\"[数据库备份告警] $message\"}}"
    fi
    # 邮件告警
    if [ -n "$ALERT_EMAIL" ]; then
        echo "$message" | mail -s "[数据库备份告警]" "$ALERT_EMAIL"
    fi
}

# 创建备份目录
mkdir -p ${BACKUP_DIR}

log "开始全量备份..."

# 执行备份
BACKUP_FILE="${BACKUP_DIR}/${DB_NAME}_full_${DATE}.sql.gz"

mysqldump -u${DB_USER} -p${DB_PASS} \
    --single-transaction \
    --routines \
    --triggers \
    --events \
    --complete-insert \
    --hex-blob \
    --quick \
    --lock-tables=false \
    ${DB_NAME} 2>> ${LOG_FILE} | gzip > ${BACKUP_FILE}

# 检查备份结果
if [ $? -eq 0 ] && [ -f ${BACKUP_FILE} ]; then
    # 计算文件大小
    FILE_SIZE=$(du -h ${BACKUP_FILE} | cut -f1)

    # 生成 MD5 校验和
    md5sum ${BACKUP_FILE} > ${BACKUP_FILE}.md5

    log "全量备份成功: ${BACKUP_FILE} (${FILE_SIZE})"

    # 上传到 OSS
    if command -v ossutil &> /dev/null; then
        ossutil cp ${BACKUP_FILE} oss://backup-bucket/mysql/ >> ${LOG_FILE} 2>&1
        ossutil cp ${BACKUP_FILE}.md5 oss://backup-bucket/mysql/ >> ${LOG_FILE} 2>&1
        log "备份已上传到 OSS"
    fi
else
    log "全量备份失败!"
    send_alert "MySQL 全量备份失败，请检查日志: ${LOG_FILE}"
    exit 1
fi

# 清理过期备份
log "清理 ${RETENTION_DAYS} 天前的备份..."
find ${BACKUP_DIR} -name "*.sql.gz" -mtime +${RETENTION_DAYS} -delete
find ${BACKUP_DIR} -name "*.md5" -mtime +${RETENTION_DAYS} -delete

log "全量备份完成"
```

### 增量备份脚本（基于 binlog）

```bash
#!/bin/bash
# /home/user/scripts/mysql_incremental_backup.sh
# MySQL 增量备份脚本

set -e

BACKUP_DIR="/data/backup/mysql/binlog"
DATE=$(date +%Y%m%d_%H%M%S)
BINLOG_DIR="/var/lib/mysql"
LOG_FILE="${BACKUP_DIR}/incremental.log"

mkdir -p ${BACKUP_DIR}

log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" >> ${LOG_FILE}
}

log "开始增量备份..."

# 刷新日志
mysql -e "FLUSH LOGS;"

# 获取当前 binlog 位置
mysql -e "SHOW MASTER STATUS\G" > ${BACKUP_DIR}/${DATE}_position.txt

# 复制 binlog 文件
BINLOG_FILES=$(ls ${BINLOG_DIR}/mysql-bin.* 2>/dev/null | grep -v index || true)

if [ -n "$BINLOG_FILES" ]; then
    mkdir -p ${BACKUP_DIR}/${DATE}
    cp ${BINLOG_DIR}/mysql-bin.* ${BACKUP_DIR}/${DATE}/

    # 压缩
    tar -czf ${BACKUP_DIR}/binlog_${DATE}.tar.gz -C ${BACKUP_DIR}/${DATE} .
    rm -rf ${BACKUP_DIR}/${DATE}

    log "增量备份成功: binlog_${DATE}.tar.gz"

    # 上传到 OSS
    if command -v ossutil &> /dev/null; then
        ossutil cp ${BACKUP_DIR}/binlog_${DATE}.tar.gz oss://backup-bucket/mysql/binlog/
    fi
else
    log "没有新的 binlog 文件"
fi

# 清理 7 天前的增量备份
find ${BACKUP_DIR} -name "binlog_*.tar.gz" -mtime +7 -delete

log "增量备份完成"
```

### 备份验证脚本

```bash
#!/bin/bash
# /home/user/scripts/verify_backup.sh
# 备份验证脚本

set -e

BACKUP_DIR="/data/backup/mysql"
VERIFY_DB="backup_verify_$(date +%Y%m%d)"
LATEST_BACKUP=$(ls -t ${BACKUP_DIR}/*.sql.gz 2>/dev/null | head -1)
LOG_FILE="${BACKUP_DIR}/verify.log"

log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" >> ${LOG_FILE}
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1"
}

if [ -z "$LATEST_BACKUP" ]; then
    log "错误: 没有找到备份文件"
    exit 1
fi

log "开始验证备份: ${LATEST_BACKUP}"

# 验证 MD5
if [ -f "${LATEST_BACKUP}.md5" ]; then
    if md5sum -c "${LATEST_BACKUP}.md5" > /dev/null 2>&1; then
        log "MD5 校验通过"
    else
        log "错误: MD5 校验失败"
        exit 1
    fi
fi

# 创建测试数据库
mysql -e "CREATE DATABASE IF NOT EXISTS ${VERIFY_DB}"

# 恢复备份
log "恢复备份到测试数据库..."
gunzip -c ${LATEST_BACKUP} | mysql ${VERIFY_DB}

if [ $? -eq 0 ]; then
    # 验证数据
    TABLES=$(mysql -N -e "SELECT COUNT(*) FROM information_schema.tables WHERE table_schema='${VERIFY_DB}'")
    log "验证成功: 恢复了 ${TABLES} 个表"

    # 清理测试数据库
    mysql -e "DROP DATABASE ${VERIFY_DB}"
    log "测试数据库已清理"
else
    log "错误: 备份恢复失败"
    mysql -e "DROP DATABASE IF EXISTS ${VERIFY_DB}"
    exit 1
fi

log "备份验证完成"
```

## 配置备份

### 应用配置备份脚本

```bash
#!/bin/bash
# /home/user/scripts/config_backup.sh
# 配置文件备份

set -e

BACKUP_DIR="/data/backup/config"
APP_DIR="/home/user/wechat-flowershop-backend"
DATE=$(date +%Y%m%d)

mkdir -p ${BACKUP_DIR}

# 备份应用配置
tar -czf ${BACKUP_DIR}/app_config_${DATE}.tar.gz \
    ${APP_DIR}/backend/.env \
    ${APP_DIR}/docker-compose.yml \
    ${APP_DIR}/nginx.conf \
    --ignore-failed-read 2>/dev/null

# 备份系统配置
tar -czf ${BACKUP_DIR}/sys_config_${DATE}.tar.gz \
    /etc/nginx/nginx.conf \
    /etc/nginx/sites-available/ \
    /etc/mysql/mysql.conf.d/ \
    /etc/redis/redis.conf \
    --ignore-failed-read 2>/dev/null

# 上传到 OSS
ossutil cp ${BACKUP_DIR}/app_config_${DATE}.tar.gz oss://backup-bucket/config/
ossutil cp ${BACKUP_DIR}/sys_config_${DATE}.tar.gz oss://backup-bucket/config/

# 清理 90 天前的配置备份
find ${BACKUP_DIR} -name "*.tar.gz" -mtime +90 -delete

echo "[$(date)] 配置备份完成"
```

### 主备份脚本

```bash
#!/bin/bash
# /home/user/scripts/backup_all.sh
# 主备份脚本 - 协调所有备份任务

set -e

LOG_FILE="/var/log/backup_all.log"
ALERT_EMAIL="admin@example.com"

log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a ${LOG_FILE}
}

send_report() {
    local status=$1
    local details=$2
    mail -s "[备份报告] ${status}" ${ALERT_EMAIL} <<EOF
备份报告
=========
时间: $(date '+%Y-%m-%d %H:%M:%S')
状态: ${status}

详情:
${details}

EOF
}

log "========== 开始备份流程 =========="

REPORT=""
ALL_SUCCESS=true

# 1. MySQL 全量备份
log "执行 MySQL 全量备份..."
if /home/user/scripts/mysql_full_backup.sh; then
    REPORT+="MySQL 全量备份: 成功\n"
else
    REPORT+="MySQL 全量备份: 失败\n"
    ALL_SUCCESS=false
fi

# 2. 配置备份
log "执行配置备份..."
if /home/user/scripts/config_backup.sh; then
    REPORT+="配置备份: 成功\n"
else
    REPORT+="配置备份: 失败\n"
    ALL_SUCCESS=false
fi

# 3. OSS 文件同步
log "执行 OSS 文件同步..."
if /home/user/scripts/oss_sync.sh; then
    REPORT+="OSS 文件同步: 成功\n"
else
    REPORT+="OSS 文件同步: 失败\n"
    ALL_SUCCESS=false
fi

# 4. 验证备份
log "验证备份..."
if /home/user/scripts/verify_backup.sh; then
    REPORT+="备份验证: 成功\n"
else
    REPORT+="备份验证: 失败\n"
    ALL_SUCCESS=false
fi

log "========== 备份流程结束 =========="

# 发送报告
if [ "$ALL_SUCCESS" = true ]; then
    send_report "成功" "$REPORT"
else
    send_report "部分失败" "$REPORT"
fi
```

## 数据恢复

### MySQL 全量恢复

```bash
#!/bin/bash
# /home/user/scripts/mysql_restore.sh
# MySQL 全量恢复脚本

BACKUP_FILE=$1
DB_NAME="wechat_flowershop"

if [ -z "$BACKUP_FILE" ]; then
    echo "用法: $0 <备份文件路径>"
    echo "示例: $0 /data/backup/mysql/wechat_flowershop_full_20240204.sql.gz"
    exit 1
fi

if [ ! -f "$BACKUP_FILE" ]; then
    echo "错误: 备份文件不存在: $BACKUP_FILE"
    exit 1
fi

echo "=========================================="
echo "警告: 此操作将覆盖当前数据库!"
echo "备份文件: $BACKUP_FILE"
echo "目标数据库: $DB_NAME"
echo "=========================================="
echo ""
read -p "确定要继续吗? (输入 yes 继续): " confirm

if [ "$confirm" != "yes" ]; then
    echo "操作已取消"
    exit 0
fi

echo ""
echo "[$(date)] 开始恢复..."

# 停止应用服务
echo "[$(date)] 停止应用服务..."
pm2 stop flowershop-api 2>/dev/null || true

# 验证 MD5（如果有）
if [ -f "${BACKUP_FILE}.md5" ]; then
    echo "[$(date)] 验证 MD5..."
    if ! md5sum -c "${BACKUP_FILE}.md5"; then
        echo "错误: MD5 校验失败"
        exit 1
    fi
fi

# 恢复数据库
echo "[$(date)] 恢复数据库..."
gunzip -c ${BACKUP_FILE} | mysql ${DB_NAME}

if [ $? -eq 0 ]; then
    echo "[$(date)] 数据库恢复成功"
else
    echo "[$(date)] 数据库恢复失败"
    exit 1
fi

# 启动应用服务
echo "[$(date)] 启动应用服务..."
pm2 start flowershop-api

echo "[$(date)] 恢复完成!"
echo ""
echo "请验证:"
echo "1. 检查应用是否正常运行: pm2 status"
echo "2. 检查数据是否正确: 访问 /health 端点"
```

### 基于时间点的恢复（PITR）

```bash
#!/bin/bash
# /home/user/scripts/mysql_pitr_restore.sh
# 基于时间点的恢复脚本

FULL_BACKUP=$1
TARGET_TIME=$2

if [ -z "$FULL_BACKUP" ] || [ -z "$TARGET_TIME" ]; then
    echo "用法: $0 <全量备份文件> <目标时间>"
    echo "示例: $0 /data/backup/mysql/full_20240204.sql.gz '2024-02-04 10:30:00'"
    exit 1
fi

echo "=========================================="
echo "基于时间点的恢复 (PITR)"
echo "全量备份: $FULL_BACKUP"
echo "目标时间: $TARGET_TIME"
echo "=========================================="

read -p "确定要继续吗? (输入 yes 继续): " confirm
if [ "$confirm" != "yes" ]; then
    echo "操作已取消"
    exit 0
fi

# 停止应用
pm2 stop flowershop-api 2>/dev/null || true

# Step 1: 恢复全量备份
echo "[$(date)] Step 1: 恢复全量备份..."
gunzip -c ${FULL_BACKUP} | mysql wechat_flowershop

# Step 2: 应用 binlog 到指定时间点
echo "[$(date)] Step 2: 应用 binlog..."
BINLOG_DIR="/data/backup/mysql/binlog"
for binlog in $(ls ${BINLOG_DIR}/*.tar.gz | sort); do
    tar -xzf $binlog -C /tmp/binlog_restore/
    mysqlbinlog --stop-datetime="${TARGET_TIME}" /tmp/binlog_restore/mysql-bin.* | mysql
    rm -rf /tmp/binlog_restore/*
done

# 启动应用
pm2 start flowershop-api

echo "[$(date)] PITR 恢复完成!"
```

### MySQL 主从配置

#### 主库配置 (my.cnf)

```ini
[mysqld]
server-id = 1
log_bin = mysql-bin
binlog_format = ROW
expire_logs_days = 7
sync_binlog = 1
gtid_mode = ON
enforce_gtid_consistency = ON

# 主从复制优化
binlog_cache_size = 4M
max_binlog_size = 512M
```

#### 从库配置 (my.cnf)

```ini
[mysqld]
server-id = 2
relay_log = relay-bin
read_only = 1
gtid_mode = ON
enforce_gtid_consistency = ON

# 复制过滤
replicate-do-db = wechat_flowershop
```

#### 配置复制

```sql
-- 在主库创建复制账号
CREATE USER 'replication'@'%' IDENTIFIED BY 'strong_password';
GRANT REPLICATION SLAVE ON *.* TO 'replication'@'%';
FLUSH PRIVILEGES;

-- 在从库配置主库连接
CHANGE MASTER TO
    MASTER_HOST='主库IP',
    MASTER_USER='replication',
    MASTER_PASSWORD='strong_password',
    MASTER_AUTO_POSITION=1;

START SLAVE;

-- 检查复制状态
SHOW SLAVE STATUS\G
```

## 灾难恢复流程

### 故障切换流程

```
┌─────────────────────────────────────────────────────────────────┐
│                    灾难恢复流程                                  │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│  1. 确认故障 (5分钟)                                            │
│     ├── 收到监控告警                                            │
│     ├── 确认主数据中心不可用                                    │
│     └── 启动灾难恢复流程                                        │
│                                                                 │
│  2. 切换决策 (5分钟)                                            │
│     ├── 评估故障影响                                            │
│     ├── 确认数据同步状态                                        │
│     └── 决定是否切换                                            │
│                                                                 │
│  3. 执行切换 (15分钟)                                           │
│     ├── 将从库提升为主库                                        │
│     ├── 更新 DNS 指向灾备中心                                   │
│     ├── 启动灾备中心的应用                                      │
│     └── 验证服务可用性                                          │
│                                                                 │
│  4. 通知相关方 (5分钟)                                          │
│     ├── 通知运维团队                                            │
│     ├── 通知开发团队                                            │
│     └── 通知业务方                                              │
│                                                                 │
```

### 故障切换脚本

```bash
#!/bin/bash
# /home/user/scripts/failover.sh
# 灾难恢复切换脚本

set -e

MASTER_IP="192.168.1.10"
SLAVE_IP="192.168.1.11"
VIP="192.168.1.100"

echo "=========================================="
echo "灾难恢复切换"
echo "主库: $MASTER_IP"
echo "从库: $SLAVE_IP"
echo "=========================================="

# 检查主库状态
echo "[$(date)] 检查主库状态..."
if mysql -h${MASTER_IP} -e "SELECT 1" > /dev/null 2>&1; then
    echo "主库仍然可用，是否确定要切换?"
    read -p "输入 'FAILOVER' 确认: " confirm
    if [ "$confirm" != "FAILOVER" ]; then
        echo "操作已取消"
        exit 0
    fi
fi

echo "[$(date)] 开始故障切换..."

# Step 1: 停止从库复制
echo "[$(date)] Step 1: 停止从库复制..."
mysql -h${SLAVE_IP} -e "STOP SLAVE;"
mysql -h${SLAVE_IP} -e "RESET SLAVE ALL;"

# Step 2: 将从库设为可写
echo "[$(date)] Step 2: 将从库设为可写..."
mysql -h${SLAVE_IP} -e "SET GLOBAL read_only = 0;"

# Step 3: 更新应用配置
echo "[$(date)] Step 3: 更新应用配置..."
sed -i "s/DB_HOST=.*/DB_HOST=${SLAVE_IP}/" /home/user/wechat-flowershop-backend/backend/.env

# Step 4: 重启应用
echo "[$(date)] Step 4: 重启应用..."
pm2 restart flowershop-api

# Step 5: 健康检查
echo "[$(date)] Step 5: 健康检查..."
sleep 10
if curl -s http://localhost:3000/health | grep -q "healthy"; then
    echo "[$(date)] 服务恢复正常"
else
    echo "[$(date)] 警告: 服务可能未完全恢复"
fi

echo "[$(date)] 故障切换完成!"
echo "新主库: ${SLAVE_IP}"
```

```bash
#!/bin/bash
# /home/user/scripts/dr_drill.sh
```
