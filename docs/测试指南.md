## 单元测试

### 运行单元测试

```bash
# 运行所有单元测试
npm test

# 运行特定测试文件
npm test -- tests/unit/userService.test.js

# 运行匹配特定名称的测试
npm test -- --testNamePattern="should create user"
```

### 单元测试示例

#### 服务层测试

```javascript
// tests/unit/productService.test.js
describe('ProductService', () => {
  describe('getProducts', () => {
    it('应该返回带分页的商品列表', async () => {
      // 准备测试数据
      const mockProducts = {
        rows: [{ id: 1, name: '红玫瑰' }],
        count: 1,
      };

      Product.findAndCountAll.mockResolvedValue(mockProducts);

      // 执行测试
      const result = await ProductService.getProducts(1, 10, {});

      // 验证结果
      expect(result.data).toEqual(mockProducts.rows);
      expect(result.pagination.total).toBe(1);
    });

    it('应该支持按分类筛选', async () => {
      await ProductService.getProducts(1, 10, { category_id: 1 });

      expect(Product.findAndCountAll).toHaveBeenCalledWith(
        expect.objectContaining({
          where: { category_id: 1 }
        })
      );
    });
  });
});
```

#### 工具函数测试

```javascript
// tests/unit/utils.test.js
describe('工具函数测试', () => {
  describe('generateOrderNo', () => {
    it('应该生成正确格式的订单号', () => {
      const orderNo = OrderService.generateOrderNo();

      expect(orderNo).toMatch(/^ORD\d{8}[A-Z0-9]{6}$/);
    });

    it('生成的订单号应该唯一', () => {
      const orderNo1 = OrderService.generateOrderNo();
      const orderNo2 = OrderService.generateOrderNo();

      expect(orderNo1).not.toBe(orderNo2);
    });
  });
});
```

## 集成测试

### 运行集成测试

```bash
npm test -- tests/api.test.js
```

### 集成测试示例

```javascript
// tests/api.test.js
const request = require('supertest');
const app = require('../server');

describe('API 集成测试', () => {
  let authToken;

  beforeAll(async () => {
    // 获取测试用的认证 Token
    authToken = generateTestToken();
  });

  describe('商品 API', () => {
    it('GET /api/products 应该返回商品列表', async () => {
      const response = await request(app)
        .get('/api/products')
        .expect(200);

      expect(response.body).toHaveProperty('data');
      expect(response.body).toHaveProperty('pagination');
      expect(Array.isArray(response.body.data)).toBe(true);
    });

    it('GET /api/products/featured 应该返回推荐商品', async () => {
      const response = await request(app)
        .get('/api/products/featured')
        .expect(200);

      expect(response.body).toHaveProperty('data');
    });

    it('POST /api/products 无Token应该返回401', async () => {
      const response = await request(app)
        .post('/api/products')
        .send({ name: '测试商品', price: 99 })
        .expect(401);

      expect(response.body.error).toBeDefined();
    });
  });

  describe('认证 API', () => {
    it('POST /api/auth/wx-login 缺少code应该返回400', async () => {
      const response = await request(app)
        .post('/api/auth/wx-login')
        .send({})
        .expect(400);

      expect(response.body.error).toBeDefined();
    });
  });
});
```

## 性能测试

### 使用 Artillery 进行负载测试

#### 1. 安装 Artillery

```bash
npm install -g artillery
```

#### 2. 创建测试配置

```yaml
# artillery/load-test.yml
config:
  target: 'http://localhost:3000'
  phases:
    - duration: 60
      arrivalRate: 10
      name: "预热阶段"
    - duration: 120
      arrivalRate: 50
      name: "正常负载"
    - duration: 60
      arrivalRate: 100
      name: "高峰负载"
  defaults:
    headers:
      Content-Type: 'application/json'

scenarios:
  - name: "获取商品列表"
    weight: 40
    flow:
      - get:
          url: "/api/products"

  - name: "获取推荐商品"
    weight: 30
    flow:
      - get:
          url: "/api/products/featured"

  - name: "搜索商品"
    weight: 20
    flow:
      - get:
          url: "/api/products/search?keyword=玫瑰"

  - name: "获取商品详情"
    weight: 10
    flow:
      - get:
          url: "/api/products/1"
```

#### 3. 运行性能测试

```bash
artillery run artillery/load-test.yml
```

### 性能指标要求

| 指标 | 目标值 | 说明 |
|------|--------|------|
| 响应时间 P95 | < 200ms | 95%请求应在200ms内完成 |
| 响应时间 P99 | < 500ms | 99%请求应在500ms内完成 |
| 吞吐量 | > 500 req/s | 每秒处理请求数 |
| 错误率 | < 0.1% | 错误请求比例 |
| 并发连接 | > 1000 | 支持的并发连接数 |

## 安全测试

### SQL 注入测试

```javascript
describe('SQL注入防护', () => {
  it('搜索接口应该防止SQL注入', async () => {
    const maliciousInput = "'; DROP TABLE users; --";

    const response = await request(app)
      .get(`/api/products/search?keyword=${encodeURIComponent(maliciousInput)}`)
      .expect(200);

    // 应该正常返回，不应该导致数据库错误
    expect(response.body.data).toBeDefined();
  });
});
```

### XSS 测试

```javascript
describe('XSS防护', () => {
  it('用户输入应该被正确处理', async () => {
    const xssPayload = '<script>alert("xss")</script>';

    const response = await request(app)
      .put('/api/users/profile')
      .set('Authorization', `Bearer ${userToken}`)
      .send({ nickname: xssPayload })
      .expect(200);

    // 验证返回的数据不包含原始脚本标签
  });
});
```

### 认证测试

```javascript
describe('认证安全', () => {
  it('过期Token应该被拒绝', async () => {
    const expiredToken = jwt.sign(
      { id: 1, type: 'user' },
      process.env.JWT_SECRET,
      { expiresIn: '-1h' }
    );

    const response = await request(app)
      .get('/api/users/profile')
      .set('Authorization', `Bearer ${expiredToken}`)
      .expect(401);

    expect(response.body.error).toContain('expired');
  });

  it('无效Token应该被拒绝', async () => {
    const response = await request(app)
      .get('/api/users/profile')
      .set('Authorization', 'Bearer invalid_token')
      .expect(403);

    expect(response.body.error).toBeDefined();
  });
});
```

## 持续集成

### GitHub Actions 配置

```yaml
# .github/workflows/test.yml
name: 测试流程

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

jobs:
  test:
    runs-on: ubuntu-latest

    services:
      mysql:
        image: mysql:8.0
        env:
          MYSQL_ROOT_PASSWORD: root
          MYSQL_DATABASE: test_db
        ports:
          - 3306:3306
        options: >-
          --health-cmd="mysqladmin ping"
          --health-interval=10s
          --health-timeout=5s
          --health-retries=3

    steps:
      - uses: actions/checkout@v4

      - name: 设置 Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'
          cache-dependency-path: backend/package-lock.json

      - name: 安装依赖
        run: cd backend && npm ci

      - name: 代码检查
        run: cd backend && npm run lint

      - name: 运行测试
        run: cd backend && npm run test:coverage

      - name: 上传覆盖率报告
        uses: codecov/codecov-action@v3
        with:
          file: ./backend/coverage/lcov.info
          fail_ci_if_error: true
```
